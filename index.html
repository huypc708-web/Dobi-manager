// Ví dụ logic chuẩn cho file API verify trên Vercel (Node.js) import { createClient } from '@supabase/supabase-js' const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_ANON_KEY) export default async function handler(req, res) { const { key, hwid } = req.query; if (!key) { return res.status(400).json({ message: "Thiếu Key!" }); } // 1. Truy vấn vào bảng chứa key trong Supabase xem key có tồn tại không const { data, error } = await supabase .from('ten_bang_key_cua_ban') // Thay tên bảng của bạn vào đây .select('*') .eq('key', key); // 2. Nếu không tìm thấy key trong database -> Bắt buộc trả về lỗi (Không cho active) if (error || !data || data.length === 0) { return res.status(404).json({ message: "Key không tồn tại trên hệ thống!" }); } const keyData = data[0]; // 3. Kiểm tra trạng thái của Key if (keyData.status !== 'active') { return res.status(400).json({ message: "Key đã bị khóa hoặc hết hạn!", status: keyData.status }); } // Nếu mọi thứ hợp lệ mới trả về thành công return res.status(200).json({ message: "Ket noi Supabase thanh cong!", total_keys: 1, data: [keyData] }); }
